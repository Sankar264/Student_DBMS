mysql> -- Q40: Safely enroll a student in a course using a TRANSACTION
mysql> -- A transaction groups multiple statements so they either ALL succeed
mysql> -- together (COMMIT) or ALL get undone together (ROLLBACK) if something
mysql> -- goes wrong midway — the database is never left half-updated.
mysql> START TRANSACTION;
Query OK, 0 rows affected (0.00 sec)

mysql>  
mysql>   INSERT INTO Enrollment (StudentID, CourseID, Semester, Grade, EnrollmentDate)
    ->   VALUES (6, 201, 'Sem2', NULL, CURDATE());
Query OK, 1 row affected (0.00 sec)

mysql>   -- Sneha Gupta (StudentID 6) is now enrolled in Database Systems for Sem2
mysql>  
mysql>   -- If this were part of a bigger booking flow (e.g. also decreasing a
mysql>   -- seat-count column), it would go here in the SAME transaction, so
mysql>   -- either both changes happen or neither does.
mysql>  
mysql> COMMIT;
Query OK, 0 rows affected (0.00 sec)

mysql> -- The change is now permanent. If anything had looked wrong before this
mysql> -- line, running ROLLBACK; instead of COMMIT; would undo the INSERT above
mysql> -- completely, as if it never happened.
mysql>
